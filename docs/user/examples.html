Enterprise Cookbook Recipes

Real-world JSON-to-DOM templates for form fields, cards, badges, and interactive controls.

Recipe 1: Form Input Field with Validation State

Binds dynamic labels, IDs, validation feedback, and placeholder text in a single pass:

import applyTemplate from "json-transformer-funcs-primitive";

const fieldData = {
  fieldId: "user-email",
  fieldName: "email",
  fieldLabel: "Corporate Email Address",
  placeholderText: "name@company.com",
  inputClass: "form-control is-valid",
  feedbackClass: "valid-feedback",
  feedbackText: "Looks good!"
};

const fieldTemplate = {
  tagName: "div",
  attributes: { class: "mb-3" },
  children: [
    {
      tagName: "label",
      attributes: { for: "${fieldId}", class: "form-label fw-semibold" },
      textContent: "${fieldLabel}"
    },
    {
      tagName: "input",
      attributes: {
        type: "email",
        id: "${fieldId}",
        name: "${fieldName}",
        class: "${inputClass}",
        placeholder: "${placeholderText}"
      }
    },
    {
      tagName: "div",
      attributes: { class: "${feedbackClass}" },
      textContent: "${feedbackText}"
    }
  ]
};

const result = applyTemplate(fieldData, fieldTemplate);

Recipe 2: Dynamic Action Button Group

Demonstrates how one data dictionary simultaneously themes primary and secondary actions:

const buttonData = {
  wrapperClass: "btn-group shadow-sm",
  primaryClass: "btn btn-primary px-4",
  secondaryClass: "btn btn-outline-secondary",
  primaryAction: "Save Contact",
  secondaryAction: "Discard"
};

const buttonTemplate = {
  tagName: "div",
  attributes: { class: "${wrapperClass}", role: "group" },
  children: [
    {
      tagName: "button",
      attributes: { type: "submit", class: "${primaryClass}" },
      textContent: "${primaryAction}"
    },
    {
      tagName: "button",
      attributes: { type: "button", class: "${secondaryClass}" },
      textContent: "${secondaryAction}"
    }
  ]
};

const result = applyTemplate(buttonData, buttonTemplate);

Recipe 3: Bootstrap 5 Collapsible Accordion

Interpolates multiple coupled IDs and target references across accordion triggers and bodies:

const accordionData = {
  headingId: "headingOne",
  collapseId: "collapseOne",
  parentId: "mainAccordion",
  title: "Step 1: Account Verification",
  bodyText: "Please upload your government-issued ID to activate your account."
};

const accordionTemplate = {
  tagName: "div",
  attributes: { class: "accordion-item" },
  children: [
    {
      tagName: "h2",
      attributes: { class: "accordion-header", id: "${headingId}" },
      children: [
        {
          tagName: "button",
          attributes: {
            class: "accordion-button",
            type: "button",
            "data-bs-toggle": "collapse",
            "data-bs-target": "#${collapseId}"
          },
          textContent: "${title}"
        }
      ]
    },
    {
      tagName: "div",
      attributes: {
        id: "${collapseId}",
        class: "accordion-collapse collapse show",
        "data-bs-parent": "#${parentId}"
      },
      children: [
        {
          tagName: "div",
          attributes: { class: "accordion-body" },
          textContent: "${bodyText}"
        }
      ]
    }
  ]
};

const result = applyTemplate(accordionData, accordionTemplate);
